public int lotesEtapa(int producto, int etapa) {
    DomitecPlant p = (DomitecPlant) domitecPlant;
    if (producto == 0) {
        switch (etapa) {
            case 0: return p.queueMateriaPrima.size();
            case 1: return p.delayDilucion.size();
            case 2: return p.delayQCLavandina.size();
            case 3: return p.queueTanquePulmon.size();
            case 4: return p.delayEnvasado.size();
        }
    }
    com.anylogic.libraries.processmodeling.Queue<? extends com.anylogic.engine.Agent> tanque;
    com.anylogic.libraries.processmodeling.Delay<? extends com.anylogic.engine.Agent> envasado;
    switch(producto) {
        case 1: tanque=p.tankDesinfectante; envasado=p.delayEnvasadoDesinfectante; break;
        case 2: tanque=p.tankLavavajilla; envasado=p.delayEnvasadoLavavajilla; break;
        case 3: tanque=p.tankLavarropa; envasado=p.delayEnvasadoLavarropa; break;
        case 4: tanque=p.tankSuavizante; envasado=p.delayEnvasadoSuavizante; break;
        case 5: tanque=p.tankDetergente; envasado=p.delayEnvasadoDetergente; break;
        default: return 0;
    }
    if(etapa==3)return tanque.size();
    if(etapa==4)return envasado.size();
    int n=0;
    if(etapa==0)for(int i=0;i<p.queueEsperaElaboracion.size();i++)if(ConfiguracionDomitec.indice(p.queueEsperaElaboracion.get(i).tipo.name())==producto)n++;
    if(etapa==1)for(int i=0;i<p.delayReactoresLiquidos.size();i++)if(ConfiguracionDomitec.indice(p.delayReactoresLiquidos.get(i).tipo.name())==producto)n++;
    if(etapa==2)for(int i=0;i<p.delayQCLiquidos.size();i++)if(ConfiguracionDomitec.indice(p.delayQCLiquidos.get(i).tipo.name())==producto)n++;
    return n;
}

public void aplicarConfiguracion(ConfiguracionDomitec nueva, boolean enVivo) {
    nueva.validar();
    DomitecPlant p = (DomitecPlant) domitecPlant;
    if (enVivo && (nueva.reactores < p.delayReactoresLiquidos.size()
            || nueva.cola < p.queueEsperaElaboracion.size()
            || nueva.tanque < Math.max(p.queueTanquePulmon.size(),
                Math.max(p.tankDesinfectante.size(), Math.max(p.tankLavavajilla.size(),
                Math.max(p.tankLavarropa.size(), Math.max(p.tankSuavizante.size(),p.tankDetergente.size())))))))
        throw new IllegalArgumentException("La capacidad nueva no puede ser menor que los lotes presentes.");
    if(enVivo && nueva.maxLotesEnProceso*1000.0 < p.materiales.wip())
        throw new IllegalArgumentException("El limite de liberacion no puede ser menor que el WIP presente.");
    cfg = nueva.copiarDeep();
    factorLatigoMain = cfg.factLatigo;
    escalaTurnoMain = cfg.escalaTurno();
    p.factorLatigo = factorLatigoMain;
    p.escalaTurno = escalaTurnoMain;
    p.delayReactoresLiquidos.set_capacity(cfg.reactores);
    p.queueEsperaElaboracion.set_capacity(cfg.cola);
    p.queueTanquePulmon.set_capacity(cfg.tanque);
    p.tankDesinfectante.set_capacity(cfg.tanque);
    p.tankLavavajilla.set_capacity(cfg.tanque);
    p.tankLavarropa.set_capacity(cfg.tanque);
    p.tankSuavizante.set_capacity(cfg.tanque);
    p.tankDetergente.set_capacity(cfg.tanque);
    for (ProductState ps : p.inventorySystem) {
        int i = ConfiguracionDomitec.indice(ps.product.name());
        ps.weeklyCapacity = cfg.horas * cfg.rate[i] * cfg.capacidadFactor * cfg.mejoraEscenario / cfg.tiemposFactor;
    }
}

public void sincronizarControles() {
    uiLatigo=cfg.factLatigo;uiHoras=cfg.horas;uiMermaLav=cfg.mermaLav*100;uiMermaLiq=cfg.mermaLiq*100;
    uiTiempo=cfg.tiemposFactor;uiCapacidad=cfg.capacidadFactor*cfg.mejoraEscenario;uiWarmup=cfg.calentamiento;
    uiReactores=cfg.reactores;uiCola=cfg.cola;uiTanque=cfg.tanque;

        campoMermaLiq.setValueToDefault();campoTiempo.setValueToDefault();campoCapacidad.setValueToDefault();

    }
}

public double demandaComun(String canal, ProductType producto) {
    int c=canal.equals("Maxiconsumo")?0:canal.equals("Grandes Clientes")?1:2;
    int p=ConfiguracionDomitec.indice(producto.name());
    long semana=(long)Math.floor(time()+1e-8);
    double q=Math.round(DatosOperativos.demanda(semillaReplica,c,p,semana,cfg.medias[c][p],cfg.desvios[c][p]));
    registro.entrada(c,p,semana,q);
    if(semana>=0 && semana<100)demandaPorSemana[(int)semana]+=q;
    return q;
}

public String tablaIC(){
    ComparacionEscenarios datos=SesionEstudio.ultimoResultado;
    if(datos==null)return "Sin estudio completo.\nEjecutar las 100 ternas.\nLos IC se publican con 100 pares.\nNo se fuerza que excluyan el cero.";
    String metrica=new String[]{"fillRate","backlog","bw","stock","produccion"}[metricaIC];
    StringBuilder s=new StringBuilder("IC 95% PAREADOS | "+metrica+"\n\n");
    for(int[] par:new int[][]{{0,1},{0,2},{1,2}}){
        ComparacionEscenarios.PairCI95 c=datos.pairCI95(par[0],par[1],metrica);
        s.append("Esc ").append(par[0]).append(" - ").append(par[1]).append(" | n=").append(c.n).append("\n");
        s.append(String.format(java.util.Locale.ROOT,"Delta: %.3f\n[%.3f ; %.3f]\n",c.media,c.inferior,c.superior));
        s.append(c.inferior>0||c.superior<0?"Excluye cero\n\n":"Incluye cero\n\n");
    }return s.toString();
}

public void observarSemana(){
    ((DomitecPlant)domitecPlant).caducarPedidos();
    ((DomitecPlant)domitecPlant).materiales.verificar();
    int s=(int)Math.floor(time());
    if(s>=0&&s<100){
        registro.semanal(s,demandaPorSemana[s],ordenesPorSemana[s],produccionPorSemana[s],stockActual(),backlogActual(),wipActual());
        dsDemanda.add(time(),demandaPorSemana[s]);dsOrdenes.add(time(),ordenesPorSemana[s]);
        dsProduccionSemanal.add(time(),produccionPorSemana[s]);dsBacklog.add(time(),backlogActual());
    }
    demandaSemanalActual=0;ordenesSemanalActual=0;produccionSemanalActual=0;
}

public void validarCorridaCompleta(){
    DomitecPlant p=(DomitecPlant)domitecPlant;
    p.caducarPedidos();p.materiales.verificar();
    double demanda=registro.demandaTotal(0);
    double cierre=demanda*registro.fillRate(0)+registro.canceladoTotal(0)+registro.pendienteTotal(0);
    if(Math.abs(demanda-cierre)>1e-5 || Math.abs(backlogActual()-registro.pendienteTotal(0))>1e-5)
        throw new IllegalStateException("No reconcilia demanda = entregada + cancelada + pendiente");
    if(wipActual()>cfg.maxLotesEnProceso*1000.0+1e-5)
        throw new IllegalStateException("WIP supera las tarjetas de liberacion");
    if(time()<99.99999 || registro.numeroSemanas()!=100 || registro.numeroEntradas()!=1800)
        throw new IllegalStateException("Corrida incompleta: tiempo="+time()+", semanas="+registro.numeroSemanas()+"/100, entradas="+registro.numeroEntradas()+"/1800, semilla="+semillaReplica+", escenario="+numeroEscenario+". Ejecutar Comparacion_100x3 desde el menu Run de AnyLogic y dejarlo terminar.");
}
